local Flow = loadstring(game:HttpGet("https://raw.githubusercontent.com/acountdiscord86-art/RYEENZYXNZ/refs/heads/main/raw%20(42).txt"))()

local ui = Flow.new({
    Name = "Example UI",
    Subtitle = "RYEENZYXNZ",
    Key = Enum.KeyCode.RightShift,
    Accent = Color3.fromRGB(120, 170, 255),
    Sound = true,
    NotifyToggles = true,
})

local window = ui:Window({
    Name = "Main Window",
})

local tab = window:Tab({
    Name = "Main",
    Icon = "cube",
})

local mainCard = tab:Module({
    Name = "Player Settings",
    Desc = ".",
})

mainCard:Toggle({
    Name = "Auto Farm",
    Desc = ".",
    Default = false,
    Callback = function(state)
        print("Auto Farm:", state)
    end,
})

mainCard:Slider({
    Name = "WalkSpeed",
    Desc = "Adjust movement speed.",
    Min = 16,
    Max = 100,
    Step = 1,
    Default = 20,
    Suffix = " ws",
    Callback = function(value)
        print("WalkSpeed:", value)
        local player = game.Players.LocalPlayer
        if player and player.Character and player.Character:FindFirstChild("Humanoid") then
            player.Character.Humanoid.WalkSpeed = value
        end
    end,
})

mainCard:Dropdown({
    Name = "Select Team",
    Desc = ".",
    Options = {"Red", "Blue", "Green"},
    Default = "Red",
    Callback = function(value)
        print("Selected Team:", value)
    end,
})

mainCard:Input({
    Name = "Nickname",
    Desc = ".",
    Placeholder = "Type here...",
    Default = "",
    Callback = function(text, enterPressed)
        print("Nickname:", text, "Enter:", enterPressed)
    end,
})

mainCard:Button({
    Name = "Print Hello",
    Desc = "Simple button example.",
    Callback = function()
        print("Hello from Flow UI")
    end,
})

local miscCard = tab:Module({
    Name = "Misc",
    Desc = "Extra actions.",
})

miscCard:Buttons({
    Name = "Actions",
    Desc = "Multiple buttons in one row.",
    Buttons = {
        {
            Name = "Reset",
            Callback = function()
                print("Reset clicked")
            end,
        },
        {
            Name = "Save",
            Accent = true,
            Callback = function()
                print("Save clicked")
            end,
        }
    }
})
